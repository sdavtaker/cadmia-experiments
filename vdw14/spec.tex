\section*{UA-DEVS Model}

The tick-counter is a coupled model $N = \langle D, \{M_d\}, \text{IC},
\text{EOC}\rangle$ with $D = \{G_{1/10},\, G_1,\, K\}$.

\subsection*{Generator $G_p$ (parametric)}

\[
  G_p = \langle S_G,\, T,\, \emptyset,\, Y_G,\;
        \delta^G_{\text{int}},\, \lambda^G,\, ta^G \rangle
\]
\begin{align*}
  S_G &= \{\bullet\} \quad\text{(single state)}\\
  ta^G(\bullet) &= p \in T \quad\text{(period, a TIME value)}\\
  \delta^G_{\text{int}}(\bullet) &= \bullet \\
  \lambda^G(\bullet) &= \text{tick signal}
\end{align*}

$G_{1/10}$ uses $p = 1/10$ and outputs a tick value of $1$.
$G_1$ uses $p = 1$ and outputs a reset signal.

\subsection*{Counter $K$}

\[
  K = \langle S_K,\, T,\, X_K,\, Y_K,\;
      \delta^K_{\text{int}},\, \delta^K_{\text{ext}},\,
      \lambda^K,\, ta^K \rangle
\]
\begin{align*}
  S_K &= \mathbb{N} \times \{\text{idle},\, \text{ready}\} \\
  ta^K(n, \text{idle})  &= \infty \\
  ta^K(n, \text{ready}) &= 0 \\
  \delta^K_{\text{ext}}\!\left((n,\text{idle}),\, e,\, x\right)
    &= \begin{cases}
         (n + x,\; \text{idle})  & x \neq 0 \quad\text{(tick)}\\
         (n,\;     \text{ready}) & x = 0    \quad\text{(reset signal)}
       \end{cases}\\
  \delta^K_{\text{int}}(n, \text{ready}) &= (0,\; \text{idle}) \\
  \lambda^K(n, \text{ready}) &= n
\end{align*}

\subsection*{Coupling}

Both generators route to $K$.  The coupling notation below uses named
channels for clarity; the IA-DEVS implementation encodes the signal in the
message value (see implementation notes).

\[
  \text{IC} = \{(G_{1/10}.\texttt{out} \to K.\texttt{add}),\;
               (G_1.\texttt{out} \to K.\texttt{reset})\}
  \qquad
  \text{EOC} = \{(K.\texttt{out} \to N.\texttt{out})\}
\]

\section*{Selected Approximations}

\paragraph{TIME type.}  \texttt{decimal<3>} (millisecond resolution).
All periods in this experiment are exact multiples of $1\,\text{ms}$,
so no rounding occurs and the outward-rounding invariant is satisfied
trivially.

\paragraph{Interval kind for periods.}  Degenerate (point) intervals.
The periods $p_{G_{1/10}} = 100\,\text{ms}$ and $p_{G_1} = 1000\,\text{ms}$
are assumed known exactly.  Choosing degenerate intervals eliminates period
uncertainty from the model so that any causality error must arise from time
arithmetic rather than from model specification.

\paragraph{Interval kind for values.}  Degenerate.  Tick value $1$ and reset
signal $0$ are exact integers; both are represented as point intervals
$[1,1]$ and $[0,0]$.

% ─────────────────────────────────────────────────────────────────────────────
\section*{IA-DEVS Resulting Model}

With the approximations above, the interval counterparts of the transitions
are:

\subsection*{Generator (interval)}
\begin{align*}
  \tilde{ta}^G(\tilde{\bullet}) &= [p,\; p] \quad\text{(degenerate period)}\\
  \tilde{\delta}^G_{\text{int}}(\tilde{\bullet}) &= \tilde{\bullet} \\
  \tilde{\lambda}^G(\tilde{\bullet}) &= [v,\; v]
    \quad (v=1\text{ for }G_{1/10},\; v=0\text{ for }G_1)
\end{align*}

\subsection*{Counter $K$ (interval)}

Let $\tilde{n} \in \mathcal{I}(\mathbb{N})$.

\begin{align*}
  \tilde{ta}^K(\tilde{n},\text{idle})  &= [\infty, \infty] \\
  \tilde{ta}^K(\tilde{n},\text{ready}) &= [0, 0] \\
  \tilde{\delta}^K_{\text{ext}}\!\left((\tilde{n},\text{idle}),\,
      \tilde{e},\, \tilde{x}\right)
    &= \begin{cases}
         (\tilde{n} + \tilde{x},\; \text{idle})  & 0 \notin \tilde{x} \\
         (\tilde{n},\;             \text{ready}) & \tilde{x} = [0,0]
       \end{cases} \\
  \tilde{\delta}^K_{\text{int}}((\tilde{n},\text{ready}))
    &= ([0,0],\; \text{idle}) \\
  \tilde{\lambda}^K(\tilde{n},\text{ready})
    &= \tilde{n}
\end{align*}

\subsection*{Expected output}

Because all intervals are degenerate and arithmetic is exact under
\texttt{decimal<3>}, every internal time is a point:
\[
  \tilde{t}^{G_{1/10}}_k = [k/10,\, k/10], \qquad
  \tilde{t}^{G_1}_k      = [k,\, k]
\]
At each integer second both generators are simultaneously scheduled;
after ten tick events the counter holds:
\[
  \tilde{n} = 10 \cdot [1,1] = [10,10]
\]
$K$ always outputs the degenerate interval $[10,\,10]$.

% ─────────────────────────────────────────────────────────────────────────────
\section*{Implementation Notes (CadmIA)}

\paragraph{Portless coupling.}
UA-DEVS and IA-DEVS do not define ports; coupling is by value.  Both
generators route to $K$'s single \texttt{input\_t = int}.  The reset signal
is encoded as value $0$; any non-zero value is a tick.

\begin{center}
\begin{tabular}{@{}ll@{}}
  \toprule
  Interval value $\tilde{x}$ & Semantics for $K$ \\
  \midrule
  $[0, 0]$             & reset: output count, go to ready \\
  $\tilde{x}$, $0 \notin \tilde{x}$ & tick: add $\tilde{x}$ to accumulator \\
  \bottomrule
\end{tabular}
\end{center}

\paragraph{SELECT policy.}
At $t = [k,k]$ for integer $k$, both $G_{1/10}$ and $G_1$ are imminent.
$G_{1/10}$ must be selected first so that $K$ accumulates the $k$-th tick
before receiving the reset signal.  The \textsc{select} function is:
\[
  \textsc{select}(S) =
    \begin{cases}
      G_{1/10} & \text{if } G_{1/10} \in S \\
      G_1      & \text{if } G_1 \in S \\
      \text{first}(S) & \text{otherwise}
    \end{cases}
\]

\paragraph{Implementation status.}
Implemented and verified.  Across 100 resets, $K$ outputs $[10,10]$ at
every reset event (0 errors).  Source: \texttt{vdw14/main.cpp}.
